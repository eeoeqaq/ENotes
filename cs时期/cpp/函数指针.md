从汇编层面看，函数在内存中有自己的地址，函数指针就是指向这个地址的指针。
```c
#include <stdio.h>
// 1. 定义两个普通函数
int add(int a, int b) {
    return a + b;
}

int subtract(int a, int b) {
    return a - b;
}

int main() {
    // 2. 声明一个函数指针
    int (*operation_ptr)(int, int);

    // 3. 将函数地址赋给指针（函数名本身就代表该函数的首地址，也可以写成 &add）
    operation_ptr = add; 

    // 4. 通过指针调用函数
    // 语法1: 标准解引用调用
    int result1 = (*operation_ptr)(10, 5); 
    // 语法2: 简写调用（C语言标准允许，更推荐，因为它看起来像普通函数调用）
    int result2 = operation_ptr(10, 5);    

    printf("Add: %d\n", result2); // 输出 15

    // 5. 让指针指向另一个函数
    operation_ptr = subtract;
    printf("Subtract: %d\n", operation_ptr(10, 5)); // 输出 5
	
	// 将 "返回int，接收两个int的函数指针" 命名为 MathFuncPtr
	typedef int (*MathFuncPtr)(int, int);
	
	// 现在你可以像使用普通类型一样使用它了！
	MathFuncPtr my_ptr = add;
    return 0;
}
```
